21.3Шаг 1. Аннотации и точка расширения

package mini;

import java.lang.annotation.*;

@Retention(RetentionPolicy.RUNTIME) @Target(ElementType.TYPE)   @interface Component { }
@Retention(RetentionPolicy.RUNTIME) @Target(ElementType.FIELD)  @interface Inject { }
@Retention(RetentionPolicy.RUNTIME) @Target(ElementType.METHOD) @interface PostConstruct { }
@Retention(RetentionPolicy.RUNTIME) @Target(ElementType.METHOD) @interface Logged { }

interface BeanPostProcessor {
    default Object before(Object bean, String name) { return bean; }
    default Object after(Object bean, String name)  { return bean; }   // может вернуть другой объект — прокси
}

21.4Шаг 2. Контейнер

package mini;

import java.beans.Introspector;
import java.lang.reflect.*;
import java.util.*;
import java.util.concurrent.ConcurrentHashMap;

final class MiniContext {
    private final Map<String, Class<?>> definitions = new LinkedHashMap<>();   // имя → «описание»
    private final Map<String, Object> singletons = new ConcurrentHashMap<>();
    private final Set<String> inCreation = new LinkedHashSet<>();
    private final List<BeanPostProcessor> postProcessors = new ArrayList<>();

    MiniContext(Class<?>... components) {
        for (Class<?> c : components) register(c);     // фаза 1: только описания, объектов ещё нет
        refresh();                                      // фаза 2: экземпляры
    }

    private void register(Class<?> c) {
        if (!c.isAnnotationPresent(Component.class))
            throw new IllegalArgumentException(c.getName() + " is not a @Component");
        definitions.put(Introspector.decapitalize(c.getSimpleName()), c);
    }

    private void refresh() {
        // как registerBeanPostProcessors(): пост-процессоры создаются раньше остальных бинов
        definitions.forEach((name, type) -> {
            if (BeanPostProcessor.class.isAssignableFrom(type)) postProcessors.add((BeanPostProcessor) getBean(name));
        });
        // как preInstantiateSingletons(): создаём всё остальное
        definitions.keySet().forEach(this::getBean);
    }

    Object getBean(String name) {
        Object existing = singletons.get(name);
        if (existing != null) return existing;                                   // кэш singleton-ов
        Class<?> type = definitions.get(name);
        if (type == null) throw new IllegalStateException("No bean named '" + name + "'");
        if (!inCreation.add(name))                                               // уже создаётся выше по стеку
            throw new IllegalStateException("Circular dependency: " + inCreation + " -> " + name);
        try {
            Object bean = instantiate(type);            // createBeanInstance
            injectFields(bean);                         // populateBean
            Object exposed = initialize(bean, name);    // initializeBean — может вернуть прокси
            singletons.put(name, exposed);
            return exposed;
        } catch (ReflectiveOperationException e) {
            throw new IllegalStateException("Failed to create bean '" + name + "'", e);
        } finally {
            inCreation.remove(name);
        }
    }

    <T> T getBean(Class<T> type) {                                             // поиск по типу
        List<String> names = definitions.entrySet().stream()
                .filter(e -> type.isAssignableFrom(e.getValue()))
                .map(Map.Entry::getKey).toList();
        if (names.size() != 1)
            throw new IllegalStateException("Expected single bean of type " + type.getName() + " but found " + names);
        return type.cast(getBean(names.get(0)));
    }

    private Object instantiate(Class<?> type) throws ReflectiveOperationException {
        Constructor<?> ctor = type.getDeclaredConstructors()[0];                 // считаем, что конструктор один
        Class<?>[] paramTypes = ctor.getParameterTypes();
        Object[] args = new Object[paramTypes.length];
        for (int i = 0; i < args.length; i++) args[i] = getBean(paramTypes[i]);    // рекурсия: сначала зависимости
        ctor.setAccessible(true);
        return ctor.newInstance(args);
    }

    private void injectFields(Object bean) throws IllegalAccessException {
        for (Field f : bean.getClass().getDeclaredFields()) {
            if (f.isAnnotationPresent(Inject.class)) {
                f.setAccessible(true);
                f.set(bean, getBean(f.getType()));
            }
        }
    }

    private Object initialize(Object bean, String name) throws ReflectiveOperationException {
        Object current = bean;
        for (BeanPostProcessor pp : postProcessors) current = pp.before(current, name);
        for (Method m : bean.getClass().getDeclaredMethods()) {
            if (m.isAnnotationPresent(PostConstruct.class)) { m.setAccessible(true); m.invoke(bean); }
        }
        for (BeanPostProcessor pp : postProcessors) current = pp.after(current, name);
        return current;
    }
}
Что уже видно Порядок создания определяется зависимостями: рекурсивный getBean в instantiate создаёт сначала листья графа. Цикл в конструкторах обнаруживается множеством inCreation — и разрешить его нельзя: объекта ещё нет. @PostConstruct вызывается на исходном объекте, до after(), который создаёт прокси, — ровно как в Spring.

21.5Шаг 3. Пост-процессор с прокси

package mini;

import java.lang.reflect.*;
import java.util.Arrays;

@Component
class LoggingPostProcessor implements BeanPostProcessor {
    @Override
    public Object after(Object bean, String name) {
        Class<?> type = bean.getClass();
        boolean hasAdvice = Arrays.stream(type.getDeclaredMethods()).anyMatch(m -> m.isAnnotationPresent(Logged.class));
        if (!hasAdvice || type.getInterfaces().length == 0) return bean;         // прокси не нужен — отдаём как есть

        InvocationHandler handler = (proxy, method, args) -> {
            Method impl = type.getMethod(method.getName(), method.getParameterTypes());   // аннотация стоит на методе класса
            long t0 = System.nanoTime();
            try {
                return method.invoke(bean, args);                                  // вызов настоящего объекта (target)
            } catch (InvocationTargetException e) {
                throw e.getCause();                                                // пробрасываем исходное исключение
            } finally {
                if (impl.isAnnotationPresent(Logged.class))
                    System.out.printf("[%s.%s] %d µs%n", name, method.getName(), (System.nanoTime() - t0) / 1_000);
            }
        };
        return Proxy.newProxyInstance(type.getClassLoader(), type.getInterfaces(), handler);
    }
}

21.6Шаг 4. Запускаем

package mini;

import java.time.Instant;

interface Greeter { String greet(String who); String shout(String who); }

@Component class Clock { Instant now() { return Instant.now(); } }

@Component class Stats { int calls; }

@Component
class GreeterImpl implements Greeter {
    private final Clock clock;                        // через конструктор
    @Inject private Stats stats;                      // через поле

    GreeterImpl(Clock clock) { this.clock = clock; System.out.println("1. constructor, stats = " + stats); }

    @PostConstruct void init() { System.out.println("2. @PostConstruct, stats = " + (stats != null)); }

    @Logged public String greet(String who) { stats.calls++; return "Hello, " + who + " @ " + clock.now(); }

    public String shout(String who) { return greet(who).toUpperCase(); }     // self-invocation!
}

public class Demo {
    public static void main(String[] args) {
        MiniContext ctx = new MiniContext(LoggingPostProcessor.class, Clock.class, Stats.class, GreeterImpl.class);
        Greeter g = ctx.getBean(Greeter.class);
        System.out.println("3. " + g.getClass().getSimpleName());           // $Proxy… — не GreeterImpl
        System.out.println(g.greet("Spring"));                               // лог времени печатается
        System.out.println(g.shout("Spring"));                               // лог НЕ печатается: greet вызван через this
        // ctx.getBean(GreeterImpl.class) → ClassCastException: прокси реализует Greeter, но не наследует GreeterImpl
    }
}

На 120 строках воспроизвелись четыре «загадки» Spring: поле с @Inject равно null в конструкторе; @PostConstruct видит внедрённые поля; self-invocation обходит прокси; JDK-прокси нельзя получить по классу реализации.

21.7Упражнения: приближаем к настоящему Spring

  1. Сканирование пакета. Вместо списка классов найдите все .class в пакете через ClassLoader.getResources(path) и зарегистрируйте помеченные @Component. (В Spring — ClassPathBeanDefinitionScanner, читает байткод без загрузки классов — через ASM, а в Spring Framework 7 на Java 24+ через ClassFile API.)
  2. @Primary и неоднозначность. Сейчас два бина одного типа дают ошибку. Добавьте @Primary и выбор по имени параметра.
  3. Scope prototype. Аннотация @Prototype: не класть в singletons, создавать при каждом getBean.
  4. Ранние ссылки. Разрешите цикл через поля: после instantiate кладите «сырой» объект в карту early и отдавайте его при повторном запросе. Что сломается, если пост-процессор оборачивает бин в прокси? Это и есть причина третьего уровня кэша (тема 03).
  5. Уничтожение. Метод close(), вызывающий @PreDestroy в порядке, обратном созданию.
  6. BeanFactoryPostProcessor. Точка расширения между register и refresh, получающая definitions: сделайте процессор, который удаляет описания классов с @Profile("dev"), если системное свойство profile не равно dev.
  7. События. publish(Object event) и методы с @Listener: соберите их в initialize и вызывайте по типу параметра.
  8. CGLIB-стиль. Замените JDK-прокси на подкласс через ByteBuddy — исчезнет требование интерфейса, появится ограничение на final.

21.9Самопроверка

В мини-контейнере GreeterImpl печатает в конструкторе «stats = null». Почему?

  • Бин Stats ещё не зарегистрирован
  • Поля внедряются после создания объекта: конструктор уже отработал, когда вызывается injectFields
  • Ошибка в рефлексии
  • Stats — prototype
Чтобы записать поле, нужен существующий объект. Это общая причина, почему полевые зависимости недоступны в конструкторе и в Spring.

Почему пост-процессоры создаются в refresh() раньше остальных бинов?

  • Они легче
  • Так требует JVM
  • Бин, созданный до регистрации пост-процессора, через него не пройдёт — и не получит прокси
  • Чтобы избежать циклов
Тот же принцип, что в registerBeanPostProcessors(): сначала вся обрабатывающая инфраструктура, потом обычные бины.

Что нужно добавить, чтобы разрешить цикл A ↔ B через поля?

  • Раннюю публикацию «сырого» объекта после вызова конструктора и до внедрения полей
  • Второй поток
  • Сортировку бинов по имени
  • Ничего, это невозможно
Именно это делает addSingletonFactory в Spring; фабрика вместо готового объекта нужна, чтобы ранняя ссылка могла быть прокси.
Опишите минимальный алгоритм getBean для singleton.

Проверить кэш → найти описание → отметить «в создании» (иначе цикл) → разрешить аргументы конструктора рекурсивным getBean и создать объект → внедрить поля/сеттеры → BPP before → init-методы → BPP after (возможна подмена на прокси) → положить в кэш → снять отметку.

Почему контейнеру нужны две фазы — описания и экземпляры?

Чтобы создать бин, нужно знать обо всех остальных: кого внедрять, есть ли кандидаты, не переопределён ли бин, какие пост-процессоры применить. Сначала собирается полная картина (её ещё можно менять — BFPP, условия), и только затем создаются объекты в порядке зависимостей.

Как на рефлексии и JDK Proxy реализовать аналог @Transactional?

BeanPostProcessor в after() проверяет наличие аннотации на методах класса, создаёт Proxy.newProxyInstance с InvocationHandler: перед вызовом открыть транзакцию (connection.setAutoCommit(false), положить соединение в ThreadLocal), вызвать method.invoke(target), commit; при исключении rollback и проброс причины; в finally очистить ThreadLocal и вернуть соединение.
